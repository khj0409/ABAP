MODULE build_alv_200 OUTPUT.
IF go_container_200 IS INITIAL.

    CREATE OBJECT go_container_200
      EXPORTING container_name = 'ALV_SUMMARY'.

    CREATE OBJECT go_alv_grid_200
      EXPORTING i_parent = go_container_200.

    " 필드 카탈로그 설정
    CLEAR: gt_fieldcat_200.
    DATA: ls_fcat TYPE lvc_s_fcat.

    ls_fcat-fieldname = 'CARRID'.
    ls_fcat-coltext = '항공사'.
    APPEND ls_fcat TO gt_fieldcat_200.
    CLEAR ls_fcat.

    ls_fcat-fieldname = 'SEATSMAX_SUM'.
    ls_fcat-coltext = '총 좌석 수'.
    ls_fcat-do_sum = 'X'.
    APPEND ls_fcat TO gt_fieldcat_200.
    CLEAR ls_fcat.

    ls_fcat-fieldname = 'UTILIZATION_AVG'.
    ls_fcat-coltext = '점유율 평균(%)'.
    APPEND ls_fcat TO gt_fieldcat_200.
    CLEAR ls_fcat.

    " 레이아웃
    gs_layout_200-grid_title = '항공사별 좌석 집계'.
    gs_layout_200-zebra = 'X'.

    CALL METHOD go_alv_grid_200->set_table_for_first_display
      EXPORTING
        is_layout       = gs_layout_200
      CHANGING
        it_fieldcatalog = gt_fieldcat_200
        it_outtab       = gt_summary.

  ENDIF.
