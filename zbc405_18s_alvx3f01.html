FORM init_data .

  "SQL
  "FROM : SFLIGHT
  "FIELDS: *
  "KEY : from selection options
  "TARGET : table...
  "VERSION: 7.0

  SELECT *
    FROM sflight
    WHERE carrid IN @so_car
    AND connid IN @so_con
    INTO CORRESPONDING FIELDS OF TABLE @gt_flight.

  IF sy-subrc <> 0.

   MESSAGE e001(zbc405_18).

  ELSE.

    MESSAGE s002(zbc405_18).

  ENDIF.
* Exception LED
  LOOP AT gt_flight into gs_sflight.

    IF gs_sflight-seatsocc < 5.

       gs_sflight-light = '1'.

    ELSEIF gs_sflight-seatsocc < 100.

      gs_sflight-light  = '2'.

    else. gs_sflight-light = '3'.

    ENDIF.

     "Color.
    IF gs_sflight-planetype = '747-400'.

      gs_sflight-color = 'C610'.
    ENDIF.

    "Cell Color.
    IF gs_sflight-planetype = 'A380-800'.
      gs_colfields-fname = 'PLANETYPE'.
      gs_colfields-color-col = '7'.
      gs_colfields-color-int = '1'.
      gs_colfields-color-inv = '0'.

      APPEND gs_colfields TO gs_sflight-it_colfields.
    ENDIF.

    IF gs_sflight-fldate < sy-datum.
      gs_sflight-changes_possible = icon_space.
    ELSE.
      gs_sflight-changes_possible = icon_okay.
    ENDIF.

*light color만 update.
    MODIFY gt_flight FROM gs_sflight TRANSPORTING light
                                                   color
                                                   it_colfields
                                                   changes_possible.
   ENDLOOP.

ENDFORM.

FORM calculate_summary .
TYPES: BEGIN OF ty_group,
           carrid TYPE sflight-carrid,
           seatsmax TYPE sflight-seatsmax,
           seatsocc TYPE sflight-seatsocc,
         END OF ty_group.

  DATA: lt_temp TYPE TABLE OF ty_group,
        ls_temp TYPE ty_group.

  " 데이터 복사
  LOOP AT gt_flight INTO gs_sflight.
    ls_temp-carrid = gs_sflight-carrid.
    ls_temp-seatsmax = gs_sflight-seatsmax.
    ls_temp-seatsocc = gs_sflight-seatsocc.
    APPEND ls_temp TO lt_temp.
  ENDLOOP.

  " 집계: CARRID 기준 그룹핑
  DATA: lt_grouped TYPE SORTED TABLE OF ty_group
                   WITH UNIQUE KEY carrid,
        lv_count   TYPE i,
        lv_total_util TYPE p DECIMALS 2.

  CLEAR: gt_summary.

  LOOP AT lt_temp INTO ls_temp.
    READ TABLE gt_summary INTO gs_summary WITH KEY carrid = ls_temp-carrid.
    IF sy-subrc = 0.
      gs_summary-seatsmax_sum += ls_temp-seatsmax.
      gs_summary-utilization_avg += ( ls_temp-seatsocc * 100 ) / ls_temp-seatsmax.
      lv_count += 1.
      MODIFY gt_summary FROM gs_summary TRANSPORTING seatsmax_sum utilization_avg WHERE carrid = gs_summary-carrid.
    ELSE.
      gs_summary-carrid = ls_temp-carrid.
      gs_summary-seatsmax_sum = ls_temp-seatsmax.
      gs_summary-utilization_avg = ( ls_temp-seatsocc * 100 ) / ls_temp-seatsmax.
      APPEND gs_summary TO gt_summary.
    ENDIF.

  ENDLOOP.

ENDFORM.
